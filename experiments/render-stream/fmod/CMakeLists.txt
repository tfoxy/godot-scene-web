cmake_minimum_required(VERSION 3.20)
project(grc_fmod LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
set(CAPTURE "${CMAKE_CURRENT_SOURCE_DIR}/../capture/src")
add_library(grc_fmod SHARED src/module.cpp src/link.cpp src/lane.cpp src/take.cpp src/cache.cpp "${CAPTURE}/rs_audio_wire.cpp" src/native_api.cpp src/pck.cpp src/banks.cpp "${CAPTURE}/rs_sha256.cpp")
target_include_directories(grc_fmod PRIVATE "${CAPTURE}")
target_link_libraries(grc_fmod PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(grc_fmod PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(grc_fmod PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(UNIX AND NOT APPLE)
    target_link_options(grc_fmod PRIVATE -Wl,--no-undefined)
  endif()
endif()
install(TARGETS grc_fmod LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES grc_fmod.h DESTINATION include)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(grc_fmod_header_test test/header.c)
  target_include_directories(grc_fmod_header_test PRIVATE .)
  add_test(NAME fmod_header COMMAND grc_fmod_header_test)
  add_executable(grc_fmod_pck_test test/pck_test.cpp src/pck.cpp "${CAPTURE}/rs_sha256.cpp")
  target_include_directories(grc_fmod_pck_test PRIVATE src "${CAPTURE}")
  add_test(NAME fmod_pck COMMAND grc_fmod_pck_test "${CMAKE_CURRENT_BINARY_DIR}")
  set(RENDER_SOURCES src/take.cpp src/cache.cpp src/native_api.cpp src/banks.cpp src/pck.cpp
    "${CAPTURE}/rs_sha256.cpp" "${CAPTURE}/rs_audio_wire.cpp")
  add_executable(grc_fmod_link_test test/link_test.cpp src/link.cpp src/lane.cpp ${RENDER_SOURCES})
  target_include_directories(grc_fmod_link_test PRIVATE src "${CAPTURE}")
  target_link_libraries(grc_fmod_link_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  add_test(NAME fmod_link COMMAND grc_fmod_link_test)
  add_executable(grc_fmod_take_test test/take_test.cpp ${RENDER_SOURCES})
  target_include_directories(grc_fmod_take_test PRIVATE src "${CAPTURE}")
  target_link_libraries(grc_fmod_take_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  option(GRC_FMOD_TEST_ACCEPT_LOOP "Sabotage the test-only one-shot guard" OFF)
  if(GRC_FMOD_TEST_ACCEPT_LOOP)
    target_compile_definitions(grc_fmod_take_test PRIVATE GRC_FMOD_TEST_ACCEPT_LOOP)
  endif()
  add_test(NAME fmod_take COMMAND grc_fmod_take_test "${CMAKE_CURRENT_BINARY_DIR}")
  if(UNIX AND NOT APPLE)
    # Instrument the same module sources; the installed target has no test callbacks.
    get_target_property(MODULE_SOURCES grc_fmod SOURCES)
    add_library(grc_fmod_test_module SHARED ${MODULE_SOURCES})
    target_include_directories(grc_fmod_test_module PRIVATE "${CAPTURE}")
    target_compile_definitions(grc_fmod_test_module PRIVATE GRC_FMOD_TEST_CALLBACKS)
    target_link_libraries(grc_fmod_test_module PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(grc_fmod_test_module PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    add_executable(grc_fmod_endpoint_test test/endpoint_test.cpp "${CAPTURE}/rs_audio_wire.cpp" "${CAPTURE}/rs_sha256.cpp")
    target_include_directories(grc_fmod_endpoint_test PRIVATE . src "${CAPTURE}")
    target_link_libraries(grc_fmod_endpoint_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(grc_fmod_endpoint_test PROPERTIES ENABLE_EXPORTS YES)
    add_test(NAME fmod_endpoint_install COMMAND grc_fmod_endpoint_test "$<TARGET_FILE:grc_fmod_test_module>" install "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(fmod_endpoint_install PROPERTIES SKIP_REGULAR_EXPRESSION "SKIP native endpoint")
    add_executable(grc_fmod_render_test test/render_test.cpp src/lane.cpp ${RENDER_SOURCES})
    target_include_directories(grc_fmod_render_test PRIVATE src "${CAPTURE}")
    target_compile_definitions(grc_fmod_render_test PRIVATE GRC_FMOD_TEST_CALLBACKS)
    target_link_libraries(grc_fmod_render_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    add_test(NAME fmod_render_parity COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test/render_driver.py" "$<TARGET_FILE:grc_fmod_render_test>" "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(fmod_render_parity PROPERTIES SKIP_REGULAR_EXPRESSION "SKIP render parity")
    add_test(NAME fmod_licence_guards COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/test/test-fmod-licence.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/check-fmod-licence.sh"
      "$<TARGET_FILE:grc_fmod>" "${CMAKE_CURRENT_BINARY_DIR}")
    add_library(grc_test_fake SHARED test/fake_api.cpp)
    target_include_directories(grc_test_fake PRIVATE src)
    target_link_libraries(grc_test_fake PRIVATE ${CMAKE_DL_LIBS})
    add_executable(grc_fmod_native_test test/native_test.cpp)
    target_include_directories(grc_fmod_native_test PRIVATE .)
    target_link_libraries(grc_fmod_native_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(grc_fmod_native_test PROPERTIES ENABLE_EXPORTS YES)
    foreach(mode exact patch outside retry create bank not-loaded source-ok source-info source-open source-read source-seek source-overread source-late)
      add_test(NAME "fmod_native_${mode}" COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/test/native_driver.py"
        "$<TARGET_FILE:grc_fmod_native_test>" "$<TARGET_FILE:grc_fmod>"
        "$<TARGET_FILE:grc_test_fake>" "${CMAKE_CURRENT_BINARY_DIR}" "${mode}")
    endforeach()
    add_test(NAME fmod_endpoint_async COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/test/native_driver.py"
      "$<TARGET_FILE:grc_fmod_endpoint_test>" "$<TARGET_FILE:grc_fmod_test_module>"
      "$<TARGET_FILE:grc_test_fake>" "${CMAKE_CURRENT_BINARY_DIR}" endpoint)
    add_test(NAME fmod_native_install COMMAND grc_fmod_native_test "$<TARGET_FILE:grc_fmod>" install)
    set_tests_properties(fmod_native_install PROPERTIES SKIP_REGULAR_EXPRESSION "SKIP native install")
  endif()
endif()
